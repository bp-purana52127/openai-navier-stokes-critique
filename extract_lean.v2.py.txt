import os
import re

def extract_lean_parameters(repo_path, output_filename):
    keywords = ['force', 'forcing', 'initial', 'velocity', 'energy']
    pattern = re.compile(r'^\s*(def|structure|noncomputable def|theorem|lemma)\s+.*(' + '|'.join(keywords) + r')', re.IGNORECASE)
    lines_to_capture = 20 
    
    with open(output_filename, 'w', encoding='utf-8') as out:
        out.write("OpenAI Lean Code Extraction (Filtered: No Libraries)\n")
        out.write("="*60 + "\n\n")
        
        for root, dirs, files in os.walk(repo_path):
            # Skip dependency folders
            if '.lake' in root or '.git' in root:
                continue
                
            for file in files:
                if file.endswith('.lean'):
                    filepath = os.path.join(root, file)
                    try:
                        with open(filepath, 'r', encoding='utf-8') as f:
                            lines = f.readlines()
                            
                        for i, line in enumerate(lines):
                            if pattern.search(line):
                                out.write(f"Location: {filepath} (Line {i+1})\n")
                                out.write("-" * 40 + "\n")
                                block = lines[i : min(i + lines_to_capture, len(lines))]
                                out.writelines(block)
                                out.write("\n" + "="*60 + "\n\n")
                    except Exception as e:
                        print(f"Could not read {filepath}: {e}")

if __name__ == "__main__":
    extract_lean_parameters('./NavierStokesAndEuler', 'openai_clean_parameters.txt')
