import os
import re

# The specific files where the engineered forcing terms were identified
TARGET_FILES = [
    './NavierStokesAndEuler/Euler/TransverseMomentumRegularity.lean',
    './NavierStokesAndEuler/Euler/ContinuousAccelerationForcing.lean',
    './NavierStokesAndEuler/Euler/MeanPacketAngularForcing.lean',
    './NavierStokesAndEuler/Euler/MeanVelocityPressure.lean'
]

# The specific variables and theorems we want to trace
TARGET_TERMS = [
    'momentumForcing', 
    'forcing', 
    'angularMeanForcing', 
    'velocity_evolution'
]

def extract_full_blocks(files, terms, output_file):
    # Regex to identify the start of a new top-level Lean declaration
    decl_pattern = re.compile(r'^(def|structure|noncomputable def|theorem|lemma|instance|class)\s+')
    
    with open(output_file, 'w', encoding='utf-8') as out:
        out.write("Full Algebraic Tracing of Engineered Forcing Terms\n")
        out.write("="*80 + "\n\n")

        for filepath in files:
            if not os.path.exists(filepath):
                print(f"File not found: {filepath}")
                continue
            
            with open(filepath, 'r', encoding='utf-8') as f:
                lines = f.readlines()
            
            capturing = False
            current_block = []
            matched_term = None
            
            for line in lines:
                # Check if this line starts a new declaration block
                if decl_pattern.match(line):
                    # If we were capturing a previous block and it contained a target term, save it
                    if capturing and matched_term:
                        out.write(f"File: {filepath}\n")
                        out.write(f"Matched Term: {matched_term}\n")
                        out.write("-" * 80 + "\n")
                        out.writelines(current_block)
                        out.write("\n" + "="*80 + "\n\n")
                    
                    # Reset for the new block
                    capturing = True
                    current_block = [line]
                    matched_term = None
                    
                    # Check if the new declaration's first line contains a target term
                    for term in terms:
                        if term in line:
                            matched_term = term
                            break
                elif capturing:
                    current_block.append(line)
                    # Check if the term appears inside the body of the block
                    if not matched_term:
                        for term in terms:
                            if term in line:
                                matched_term = term
                                break
            
            # Catch the final block in the file if it matched
            if capturing and matched_term:
                out.write(f"File: {filepath}\n")
                out.write(f"Matched Term: {matched_term}\n")
                out.write("-" * 80 + "\n")
                out.writelines(current_block)
                out.write("\n" + "="*80 + "\n\n")

if __name__ == "__main__":
    output_filename = 'full_forcing_trace.txt'
    extract_full_blocks(TARGET_FILES, TARGET_TERMS, output_filename)
    print(f"Extraction complete. Check {output_filename}")
